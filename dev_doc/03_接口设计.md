# 接口设计 - GrepLogViewer

## 1. Model 层接口

### 1.1 IRegexGroupModel
```typescript
/**
 * 正则组数据模型接口
 * 负责正则组配置数据的存取和校验
 */
interface IRegexGroupModel {
  /** 创建新的正则组 */
  createGroup(name: string, color: string): RegexGroup;

  /** 向组内添加表达式 */
  addExpression(groupId: string, pattern: string, flags: string, operator: LogicOperator): RegexExpression;

  /** 删除组内表达式 */
  removeExpression(groupId: string, expressionId: string): void;

  /** 更新表达式 */
  updateExpression(groupId: string, expressionId: string, updates: Partial<RegexExpression>): void;

  /** 更新组信息（名称、颜色） */
  updateGroup(groupId: string, updates: Partial<Pick<RegexGroup, 'name' | 'color'>>): void;

  /** 删除正则组 */
  removeGroup(groupId: string): void;

  /** 获取所有组（按顺序） */
  getGroups(): RegexGroup[];

  /** 设置组顺序 */
  setGroupOrder(groupIds: string[]): void;

  /** 校验正则表达式是否合法 */
  validatePattern(pattern: string, flags: string): { valid: boolean; error?: string };
}
```

### 2.2 IFilterController
```typescript
/**
 * 过滤控制器接口
 * 负责正则匹配计算，生成过滤结果
 */
interface IFilterController {
  /**
   * 执行过滤
   * @param lines 文档所有行
   * @param groups 正则组配置
   * @returns 过滤结果（每行对应的组和颜色）
   */
  filter(lines: string[], groups: RegexGroup[]): FilterResult[];

  /**
   * 单行匹配指定正则组
   * @param line 文本行
   * @param group 正则组
   * @returns 是否匹配
   */
  matchGroup(line: string, group: RegexGroup): boolean;
}
```

### 2.3 IViewController
```typescript
/**
 * 视图控制器接口
 * 协调 ConfigPanel 和 EditorDecorations，处理编辑器生命周期
 */
interface IViewController {
  /** 初始化：为指定编辑器创建配置面板 */
  attach(editor: vscode.TextEditor): void;

  /** 分离：编辑器关闭时清理 */
  detach(editor: vscode.TextEditor): void;

  /** 处理 Go 按钮：保存配置、执行过滤、应用显示 */
  onGo(groups: RegexGroup[]): void;

  /** 处理 Reset 按钮：清除所有配置和显示效果 */
  onReset(): void;

  /** 文档变更时重新应用过滤 */
  onDocumentChange(): void;

  /** 释放所有资源 */
  dispose(): void;
}
```

## 4. 类型定义汇总

```typescript
// === 基础类型 ===

interface RegexGroup {
  id: string;
  name: string;
  color: string;
  expressions: RegexExpression[];
}

interface RegexExpression {
  id: string;
  pattern: string;
  flags: string;
  operator: LogicOperator;  // 与前一表达式的逻辑关系
}

enum LogicOperator {
  AND = 'and',
  OR = 'or',
  NOT = 'not'
}

interface FilterResult {
  lineNumber: number;
  groupId: string | null;
  color: string | undefined;
}

// === 消息协议（Webview ↔ Extension） ===

// Extension → Webview
interface UpdateConfigMessage {
  type: 'updateConfig';
  groups: RegexGroup[];
}

// Webview → Extension
interface GoMessage {
  type: 'go';
  groups: RegexGroup[];
}

interface ResetMessage {
  type: 'reset';
}

interface ValidatePatternMessage {
  type: 'validatePattern';
  pattern: string;
  flags: string;
}

interface ValidatePatternResponse {
  type: 'validatePatternResult';
  valid: boolean;
  error?: string;
}

type WebviewMessage = GoMessage | ResetMessage | ValidatePatternMessage;
type ExtensionMessage = UpdateConfigMessage | ValidatePatternResponse;
```
